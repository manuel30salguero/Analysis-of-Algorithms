cmake_minimum_required(VERSION 3.16)
project(HungarianResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(assignment src/assignment.cpp src/tracking.cpp)
target_include_directories(assignment PUBLIC include)
target_compile_options(assignment PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(assignment PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(assignment PUBLIC -fsanitize=address,undefined)
endif()
foreach(name main benchmark tracking_experiment)
  add_executable(lap_${name} src/${name}.cpp)
  target_link_libraries(lap_${name} PRIVATE assignment)
endforeach()
enable_testing()
foreach(name assignment tracking)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE assignment)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
